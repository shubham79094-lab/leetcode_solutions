// Title: Build Array from Permutation
            // Difficulty: Easy
            // Language: Python3
            // Link: https://leetcode.com/problems/build-array-from-permutation/

class Solution:
    def buildArray(self, nums: List[int]) -> List[int]:
        n = len(nums)

        for i in range(n):
            nums[i] += (nums[nums[i]] % n) * n
            
        for i in range(n):
            nums[i] //= n
        return nums
